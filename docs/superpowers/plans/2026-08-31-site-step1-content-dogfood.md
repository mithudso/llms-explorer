# LLMS-Explorer site — Step 1 (content + dog food + lint in CI) Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Ship build-order step 1 of `docs/site/00-platform-design.md` §10: the content sections (03 reference, 05 CLLMS essay, 11 V2 vs V1, 12 vocabulary intro + grammar, 14 decision table + copy-only recipes, 04's eight launch posts) as an Astro site whose every page has a `.md` twin, that publishes its own `llms.txt` / `llms-full.txt` / `llms-small.txt` / `llms-facts.txt` / `llms-vocabulary.txt`, and whose CI fails on any High lint finding — plus the `export_llms` overrides hook the master's principle 2 requires.

**Architecture:** An Astro project under `site/` (content collections for reference, essays, examples, blog) on Cloudflare Pages. Python tools under `site/tools/` (run on the vendored hub code, `hub/.venv`) generate the reference tables from the `/ldo` rubric, the figures for the blog from `outputs/**/manifest.json`, the `.md` twins + `_headers`, and the site's own llms family by writing a banner mirror of the built pages and running `docset_refine` over it. A GitHub Action builds, runs the tools, runs `pytest site/tests`, and gates on `llms_lint.py check` exit code.

**Tech Stack:** Astro 5 (Node 22), Cloudflare Pages, Python 3.12+ with the vendored hub (`hub/scripts/docset_refine`, `hub/scripts/llms_lint.py`), pytest, GitHub Actions.

## Global Constraints

- Design authority: `docs/site/00-platform-design.md` (D1 Astro; free tier content; dog food principle 6; artifact kinds §4) and `docs/site/components/{01,03,04,05,11,12,14}.md`.
- Every content page has a `.md` twin at the same route with `.md` appended (`/reference/attributes` → `/reference/attributes.md`), served `text/markdown; charset=utf-8` with `X-Markdown-Tokens` and `Link: </llms.txt>; rel="describedby"`.
- The site's own llms files are generated, never hand-edited (master principle 2); hand inputs go into `site/llms.overrides.json` (title, summary, section order).
- CI gate: `hub/.venv/bin/python hub/scripts/llms_lint.py check site/dist/llms.txt site/dist/llms-facts.txt site/dist/llms-full.txt site/dist/llms-small.txt site/dist/llms-vocabulary.txt --json` must exit 0 (no High; `--kind vocabulary` from Task 2b). `--check-links` only on `main` (network).
- Facts line grammar: `- [type] text — url#anchor` with `type ∈ UNIT_TYPES` (concept, fact, actionable, question, problem, statement, quote, idea, snippet, parameter, definition, change).
- Index ≤ 10 KB or hub-and-spoke split (`export_llms.INDEX_SPLIT_BYTES = 10_000`); small ≤ 200 000 chars.
- No model tokens in step 1: `docset_refine` runs `clean → extract → render → export` only (no `units`, no `polish`).
- Hub code changes (Task 2) are made in `~/.global-ai-hub`, committed there, and reach this repo through `scripts/refresh_snapshot.sh`; never edit `hub/` copies directly.
- Python: `hub/.venv` created by `sh hub/bootstrap.sh`; all site tools run as `hub/.venv/bin/python site/tools/<tool>.py`.
- Commits: conventional prefixes (`feat:`, `docs:`, `test:`, `ci:`); one commit per task.

def load_overrides(mirror: Path) -> dict:
    """Hand inputs the generator honours across regenerations (master principle 2):
    <stem>.llms.overrides.json beside the mirror, else manifest.json["overrides"]."""
    out_dir = mirror.parent / f"{mirror.stem}.llms"
    for cand in (mirror.parent / f"{mirror.stem}.llms.overrides.json", out_dir / "manifest.json"):
        data = mirror_io.load_json(cand, default=None)
        if not data:
            continue
        ov = data if cand.name.endswith(".overrides.json") else data.get("overrides")
        if ov:
            return {k: v for k, v in ov.items() if k in ("title", "summary", "section_order", "note")}
    return {}


def _order_sections(sections: "OrderedDict[str, list]", order: list[str] | None):
    if not order:
        return sections
    ranked = OrderedDict((n, sections[n]) for n in order if n in sections)
    for n, v in sections.items():
        ranked.setdefault(n, v)
    return ranked
```

#!/usr/bin/env python3
"""gen_reference — reference pages generated from the /ldo rubric and the /dr spokes.
Usage: gen_reference.py [--out site/src/content/reference]"""
from __future__ import annotations
import argparse, re
from pathlib import Path

SOURCES = [  # (out name, source path, title, description, order)
    ("attributes.md", "skills/llms-deep-optimizer/references/attributes.md", "The attribute rubric", "Every attribute an llms file is judged on, with bars and severities.", 20),
    ("passes.md", "skills/llms-deep-optimizer/references/passes.md", "The passes", "What the optimizer runs, in order, and how each pass is judged and fixed.", 21),
    ("spec.md", "skills/document-formats/references/llms-txt.md", "llms.txt: the spec and its grammars", "Spec v2, llms-full grammars, discovery, consumers.", 10),
    ("tooling.md", "skills/document-formats/references/llms-txt-generation-tooling.md", "Generation tooling", "Generators compared; why extractive descriptions win.", 30),
    ("evidence.md", "skills/document-formats/references/llms-txt-ecosystem-evidence.md", "Ecosystem evidence", "Who reads these files, measured.", 31),
    ("recreation.md", "skills/document-formats/references/llms-txt-recreation-and-aggregation.md", "Recreating and aggregating", "The acquisition ladder, lenient parsing, families, rights.", 32),
]
FM_RE = re.compile(r"\A---\n.*?\n---\n", re.S)


def _strip_frontmatter(text: str) -> str:
    return FM_RE.sub("", text, count=1)


def render(source_text: str, title: str, description: str, order: int, source_rel: str) -> str:
    body = _strip_frontmatter(source_text).lstrip("\n")
    fm = (f"---\ntitle: {title!r}\ndescription: {description!r}\nsection: reference\norder: {order}\n"
          f"sources:\n  - {source_rel}\n---\n\n")
    return fm + body


def generate(repo_root: Path, out_dir: Path) -> list[Path]:
    out_dir.mkdir(parents=True, exist_ok=True)
    written = []
    for name, rel, title, desc, order in SOURCES:
        src = repo_root / rel
        (out_dir / name).write_text(render(src.read_text(encoding="utf-8"), title, desc, order, rel), encoding="utf-8")
        written.append(out_dir / name)
    return written


def main(argv=None) -> int:
    p = argparse.ArgumentParser(description=__doc__)
    p.add_argument("--out", default="site/src/content/reference")
    a = p.parse_args(argv)
    root = Path(__file__).resolve().parents[2]
    for f in generate(root, root / a.out):
        print(f)
    return 0


if __name__ == "__main__":
    raise SystemExit(main())
```

```python
#!/usr/bin/env python3
"""twins — .md twins for every built page + the Cloudflare _headers file.
Usage: twins.py [--content src/content] [--dist dist] [--site-url URL]"""
from __future__ import annotations
import argparse, datetime, re
from pathlib import Path

FM_RE = re.compile(r"\A---\n(.*?)\n---\n", re.S)


def _title(fm: str) -> str:
    m = re.search(r"^title:\s*(.+)$", fm, re.M)
    return m.group(1).strip().strip("'\"") if m else "Untitled"


def write_twins(content_dir: Path, dist_dir: Path, site_url: str) -> list[Path]:
    out = []
    stamp = datetime.date.today().isoformat()
    for src in sorted(content_dir.rglob("*.md")):
        rel = src.relative_to(content_dir)                      # essays/a.md
        route = f"/{rel.with_suffix('').as_posix()}/"
        text = src.read_text(encoding="utf-8")
        m = FM_RE.match(text)
        fm, body = (m.group(1), text[m.end():]) if m else ("", text)
        twin = dist_dir / rel
        twin.parent.mkdir(parents=True, exist_ok=True)
        twin.write_text(f"<!-- llms-explorer twin of {site_url}{route} · generated {stamp} -->\n\n"
                        f"# {_title(fm)}\n\n{body.lstrip()}", encoding="utf-8")
        out.append(twin)
    return out


def write_headers(dist_dir: Path) -> Path:
    lines = ["/*.md", "  Content-Type: text/markdown; charset=utf-8", '  Link: </llms.txt>; rel="describedby"',
             "/llms*.txt", "  Content-Type: text/markdown; charset=utf-8", '  Link: </llms.txt>; rel="describedby"']
    for f in sorted(dist_dir.rglob("*.md")):
        lines += [f"/{f.relative_to(dist_dir).as_posix()}", f"  X-Markdown-Tokens: {f.stat().st_size // 4}"]
    for f in sorted(dist_dir.glob("llms*.txt")):
        lines += [f"/{f.name}", f"  X-Markdown-Tokens: {f.stat().st_size // 4}"]
    dest = dist_dir / "_headers"
    dest.write_text("\n".join(lines) + "\n", encoding="utf-8")
    return dest


def main(argv=None) -> int:
    p = argparse.ArgumentParser(description=__doc__)
    p.add_argument("--content", default="src/content"); p.add_argument("--dist", default="dist")
    p.add_argument("--site-url", default="https://llms-explorer.pages.dev")
    a = p.parse_args(argv)
    here = Path(__file__).resolve().parents[1]
    n = len(write_twins(here / a.content, here / a.dist, a.site_url.rstrip("/")))
    write_headers(here / a.dist)
    print(f"{n} twins, _headers written"); return 0


if __name__ == "__main__":
    raise SystemExit(main())
```

#!/usr/bin/env python3
"""build_llms — the site's own llms family, generated from its .md twins.
Usage: build_llms.py [--dist dist] [--site-url URL] [--work .llms-work]"""
from __future__ import annotations
import argparse, json, shutil, sys
from pathlib import Path

HERE = Path(__file__).resolve().parents[1]
sys.path.insert(0, str(HERE.parent / "hub" / "scripts"))
import llms_lint  # noqa: E402
import twins  # noqa: E402
from docset_refine import clean, export_llms, extract, render, vocabulary  # noqa: E402

BANNER = "=========="
FILES = ("llms.txt", "llms-full.txt", "llms-small.txt", "llms-facts.txt", "manifest.json")


def write_mirror(dist: Path, site_url: str, out: Path) -> int:
    parts = []
    for twin in sorted(dist.rglob("*.md")):
        if twin.name.startswith("llms"):
            continue
        route = "/" + twin.relative_to(dist).with_suffix("").as_posix() + "/"
        body = twin.read_text(encoding="utf-8").split("-->\n", 1)[-1].lstrip()
        parts.append(f"{BANNER}\nURL: {site_url}{route}\n{BANNER}\n{body}\n")
    out.write_text("".join(parts), encoding="utf-8")
    return len(parts)


def build(dist: Path, site_url: str, work: Path) -> dict:
    work.mkdir(parents=True, exist_ok=True)
    mirror = work / "site.md"
    write_mirror(dist, site_url, mirror)
    ov_src = HERE / "llms.overrides.json"
    if ov_src.exists() and not (work / "site.llms.overrides.json").exists():
        shutil.copy(ov_src, work / "site.llms.overrides.json")
    clean.run(mirror); extract.run(mirror); render.run(mirror)
    export_llms.run(mirror)
    out_dir = work / "site.llms"
    for name in FILES:
        if (out_dir / name).exists():
            shutil.copy(out_dir / name, dist / name)
    for spoke in out_dir.rglob("*/llms.txt"):                     # split sections, if any
        dest = dist / spoke.relative_to(out_dir); dest.parent.mkdir(parents=True, exist_ok=True); shutil.copy(spoke, dest)
    units = work / "site.reference" / "all_units.jsonl"
    if units.exists() and units.stat().st_size:
        # signature: run(pool_paths, subject, out_dir, tree=None, llm=None, register=False, ...)
        # tree=None loads concept-tree/tree.json via concept_tree.ConceptTree.load() — pass the

        import concept_tree as ct
        tree = ct.ConceptTree.load(HERE.parent / "concept-tree" / "tree.json")
        vocabulary.run([units], "LLMS-Explorer", out_dir, tree=tree, llm=None, log=lambda *_: None)
        if (out_dir / "llms-vocabulary.txt").exists():
            shutil.copy(out_dir / "llms-vocabulary.txt", dist / "llms-vocabulary.txt")
    twins.write_headers(dist)
    counts = {"high": 0, "medium": 0}
    for f in [dist / n for n in FILES if n != "manifest.json"] + [dist / "llms-vocabulary.txt"] + list(dist.rglob("*/llms.txt")):
        if f.exists():

def main(argv=None) -> int:
